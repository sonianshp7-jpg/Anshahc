function doPost(e) {
  // Attached Google Sheet: https://docs.google.com/spreadsheets/d/1o6IneSEj2uB5bY1zF4ZEJG3XUwIAJQMS9IU7Uty03Kk/edit
  var sheet = SpreadsheetApp.openById("1o6IneSEj2uB5bY1zF4ZEJG3XUwIAJQMS9IU7Uty03Kk").getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  if (sheet.getLastRow() === 0) {
    sheet.appendRow(["Timestamp", "Booking ID", "Client Name", "Phone", "Company / Firm", "Service Required", "Article & Weight", "Preferred Date", "Remarks / Notes"]);
    sheet.getRange(1, 1, 1, 9).setFontWeight("bold").setBackground("#d4af37");
  }
  sheet.appendRow([new Date(), data.id, data.name, data.phone, data.company, data.service, data.metal, data.date, data.notes]);
  return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
}
